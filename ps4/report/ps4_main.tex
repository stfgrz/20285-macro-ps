\section{Optimal price setting in the Taylor model} 

    In this exercise, we explore the Taylor model of staggered price setting, where each firm resets its price for \( N \) periods. In each period, a fraction \( \frac{1}{N} \) of firms reset their prices, while the remaining firms keep their prices fixed. 
    Each firm \( i \) produces a differentiated good with technology:
    \[
    Y_t(i) = A_t N_t(i),
    \]
    where \( A_t \) represents the level of technology, evolving exogenously as \( a_t = \log A_t \) with a stationary stochastic process. A firm resetting its price \( P_t^* \) in period \( t \) seeks to maximize its expected profits over \( N \) periods:
    
    \[
    \max_{P_t^*} \sum_{k=0}^{N-1} \mathbb{E}_t \left\{ Q_{t,t+k} \left( \frac{1}{P_{t+k}} \right) \left( P_t^* Y_{t+k|t} - \Psi_{t+k}(Y_{t+k|t}) \right) \right\}
    \]
    subject to:
    \[
    Y_{t+k|t} = \left( \frac{P_t^*}{P_{t+k}} \right)^{-\epsilon} C_{t+k},
    \]
    where \( Q_{t,t+k} = \beta^k \left( \frac{C_{t+k}}{C_t} \right)^{-\sigma} \) is the stochastic discount factor, $ \Psi_{t+k}(Y_{t+k|t})$ are the nominal cost, and \( C_{t+k} \) is aggregate consumption in period \( t+k \).
    
    \begin{enumerate}
        \item Show that the optimal reset price \( P_t^* \) must satisfy the first-order condition:
        \[
        \sum_{k=0}^{N-1} \mathbb{E}_t \left\{ \left( \frac{1}{P_{t+k}} \right) Q_{t,t+k} Y_{t+k|t} \left( P_t^* - \mathcal{M} \psi_{t+k} \right) \right\} = 0,
        \]
        where \( \psi_{t+k} = \Psi'(\cdot) \) is the nominal marginal cost and \( \mathcal{M} = \frac{\epsilon}{\epsilon - 1} \).
    
        \textit{[Hint: Set up the first-order condition by differentiating the profit function with respect to \( P_t^* \). Remember to substitute the demand function and use the stochastic discount factor for nominal payoffs.]}
    
            \begin{solution}
Substituting the demand function in the profit function, we get:
\begin{align*}
    \max_{P_t^*} \sum_{k=0}^{N-1} \mathbb{E}_t \left\{ Q_{t,t+k} \left( \frac{1}{P_{t+k}} \right) \left( P_t^* \left( \frac{P_t^*}{P_{t+k}} \right)^{-\epsilon} C_{t+k} - \Psi_{t+k}\left(\left( \frac{P_t^*}{P_{t+k}} \right)^{-\epsilon} C_{t+k}\right) \right) \right\}.
\end{align*}
Now, we can take the first order condition with respect to $P_t^*$:
\begin{align*}
    \sum_{k=0}^{N-1} \mathbb{E}_t \left \{ Q_{t,t+k} \left( \frac{1}{P_{t+k}} \right) \left( P_{t+k}^\varepsilon (1-\varepsilon) (P_t^*)^{-\varepsilon}C_{t+k}- \psi_{t+k} P_{t+k}^\varepsilon (-\varepsilon) (P_t^*)^{-\varepsilon-1} C_{t+k} \right)\right \}=0.
\end{align*}
Rearranging the expression and substituting back the demand expression:
\begin{align*}
     \sum_{k=0}^{N-1} \mathbb{E}_t \left \{ Q_{t,t+k} \left( \frac{1}{P_{t+k}} \right) \left((1-\varepsilon)Y_{t+k|t} + \varepsilon \psi_{t+k} \frac{1}{P_t^*} Y_{t+k|t} \right)\right \}=0  \\
     \sum_{k=0}^{N-1} \mathbb{E}_t \left \{  \left( \frac{1}{P_{t+k}} \right) Q_{t,t+k} Y_{t+k|t}\left((1-\varepsilon)P_t^* + \varepsilon \psi_{t+k} \right)\right \}=0 \\
     \sum_{k=0}^{N-1} \mathbb{E}_t \left\{ \left( \frac{1}{P_{t+k}} \right) Q_{t,t+k} Y_{t+k|t} \left( P_t^* - \mathcal{M} \psi_{t+k} \right) \right\} = 0,
\end{align*}
where, in the second line, we multiplied both sides by $P_t^*$, while in the third line we divided both sides by $1-\varepsilon$ and defined $\mathcal{M}=\frac{\varepsilon}{\varepsilon-1}$.
                
                
            \end{solution}
    
        \item Derive the log-linearized version of the optimal price-setting rule around a zero-inflation steady state:
        \[
        p_t^* = \mu + \sum_{k=0}^{N-1} \omega_k \mathbb{E}_t \left\{ \varphi_{t+k} \right\},
        \]
        where \( \omega_k = \frac{\beta^k (1 - \beta)}{1 - \beta^N} \), \( \varphi_{t+k} = \log \psi_{t+k} \), and \( \mu = \log \mathcal{M} \).
    
        Interpret this condition.
    
        \textit{[Hint: Approximate the optimal price around a zero-inflation steady state by log-linearizing the price-setting rule. This will involve taking logarithms and expectations.]}
    
            \begin{solution}
First, notice that, in a steady state with zero inflation, output and prices are constant, and the price setting rule gives:
\begin{align*}
    \sum_{k=0}^{N-1} \frac{1}{P} \beta^k \frac{u'(C)}{u'(C)} Y (P- \mathcal{M} \psi) =0,
\end{align*}
which implies:
\[
\frac{Y}{P}(P-\mathcal{M}\psi) \sum_{k=0}^{N-1}\beta ^k=0.
\]
We can get rid of the first factor and solve the geometric sum:
\[
(P-\mathcal{M}\psi) \frac{1-\beta^{N}}{1-\beta}=0 \implies P=\mathcal{M}\psi.
\]
Now, notice that we can express each variable as a function of its log-deviation from the steady state:
\[
\hat x_t= \log X_t - \log X \implies X_t=Xe^{\hat x_t},
\]
hence:
\begin{align*}
    \sum_{k=0}^{N-1} \mathbb{E}_t \left \{ \beta ^k \frac{Y}{P} e^{\hat q_{t,t+k}+\hat y_{t+k|t}-\hat p_{t+k}}(Pe^{\hat p_t^*}-\mathcal{M}\psi e^{\hat \psi _{t+k}})\right \}=0.
\end{align*}
We can expand the last expression to get:
\begin{align*}
    \sum_{k=0}^{N-1} \mathbb{E}_t \left \{ \beta ^k Y e^{\hat q_{t,t+k}+\hat y_{t+k|t}-\hat p_{t+k}+\hat p_t^*} \right \} = \sum_{k=0}^{N-1} \mathbb{E}_t \left \{ \beta^k \frac{Y}{P}\mathcal{M}\psi e^{\hat q_{t,t+k}+\hat y_{t+k|t}-\hat p_{t+k}+\hat \psi_{t+k}}  \right\}.
\end{align*}
Recalling that $P=\mathcal{M}\psi$, the expression simplifies to:
\begin{align*}
    \sum_{k=0}^{N-1} \mathbb{E}_t \left \{ \beta ^k e^{\hat q_{t,t+k}+\hat y_{t+k|t}-\hat p_{t+k}+\hat p_t^*} \right \} = \sum_{k=0}^{N-1} \mathbb{E}_t \left \{ \beta^k e^{\hat q_{t,t+k}+\hat y_{t+k|t}-\hat p_{t+k}+\hat \psi_{t+k}}  \right\}.
\end{align*}
              Now, we can take the first order Taylor expension around the steady state $\hat x=0$:
              \begin{align*}
                  \sum_{k=0}^{N-1} \mathbb{E}_t \{ \beta^k(1+\hat q_{t,t+k}+\hat y_{t+k|t}-\hat p_{t+k}+\hat p_t^*)\} \approx \sum_{k=0}^{N-1}\mathbb{E}_t \{ \beta^k (1+\hat q_{t,t+k}+\hat y_{t+k|t}-\hat p_{t+k}+\hat \psi_{t+k}).
              \end{align*}
Taking everything to the left-hand side, we have:
\[
\sum_{k=0}^{N-1} \mathbb{E}_t \{ \beta^k(\hat p_t^*-\hat \psi_{t+k}) \} =0.
\]
We can split again the sum and simplify:
\begin{align*}
    \hat p_t^*\sum_{k=0}^{N-1} \beta^k- \sum_{k=0}^{N-1} \beta ^k \mathbb{E}_t [\hat \psi_{t+k}]=0.
\end{align*}
Now, we know that $\hat p_t^* = p_t^*-p$ and $\hat \psi_t= \varphi_t-\varphi$, therefore:
\begin{align*}
    (p_t^*-p) \frac{1-\beta^N}{1-\beta}=\sum_{k=0}^{N-1}\beta^k \mathbb{E}_t [\varphi_{t+k}-\varphi]=\sum_{k=0}^{N-1}\beta^k \mathbb{E}_t [\varphi_{t+k}]-\varphi \frac{1-\beta^N}{1-\beta} .
\end{align*}
Moreover, from the steady state relation, we know that:
\[
p= \log P=\log \mathcal{M} \psi =\log \mathcal{M}+\log \psi =\mu +\varphi,
\]
and so:
\begin{align*}
    p_t^*&=\sum_{k=0}^{N-1}\frac{(1-\beta)\beta^k}{1-\beta^N}\mathbb{E}_t[\varphi_{t+k}]+p-\varphi \\
    & = \mu +\sum_{k=0}^{N-1} \omega _k \mathbb{E}_t [\varphi_{t+k}].
\end{align*}
Therefore, with the Taylor model of staggered price setting, the price $p_t^*$ set by each firm depends positively on the mark-up under flexible prices $\mathcal{M}$, on the current nominal marginal cost $\varphi_t$, and on the (discounted) expected nominal marginal costs for the following $N-1$ periods, in which the firm will not be able to reset the price.
            \end{solution}
    
        \item How does the Taylor model’s price-setting rule differ from the one generated by the Calvo model? Discuss the economic intuition behind these differences, focusing on the implications for inflation persistence. How do they compare in terms of tractability? 
    
            \begin{solution}
The Taylor and Calvo models generate similar price-setting rules in the sense that, in both cases, the reset price is a markup over a weighted average of current and expected future nominal marginal costs. The key difference is how long the newly chosen price is expected to remain in place.

In the Taylor model, a firm that resets at time $t$ knows with certainty that its price will remain fixed for exactly $N$ periods. For this reason, its optimal reset price depends on a finite weighted average of marginal costs over the next $N$ periods:
\[
p_t^*=\mu+\sum_{k=0}^{N-1}\omega_k E_t[\varphi_{t+k}].
\]
The horizon is finite because the firm knows exactly when it will be allowed to re-optimise.

In the Calvo model, instead, each firm can reset its price in any given period only with probability $1-\theta$. The duration of a price spell is therefore random. A firm resetting at time $t$ must take into account that the price may remain in place for many periods, so the optimal reset price depends on an infinite discounted sum of expected future marginal costs:
\[
p_t^*=\mu+(1-\beta\theta)\sum_{k=0}^{\infty}(\beta\theta)^k E_t[\varphi_{t+k}].
\]

Under Taylor pricing, firms face no uncertainty about the length of the price spell, so they choose a price that is optimal over a fixed horizon. Under Calvo pricing, firms do not know when they will next be able to adjust, so they place weight on marginal costs arbitrarily far in the future, although with geometrically declining weights.

This difference also matters for inflation persistence. In the Calvo model, because some firms may remain stuck with the same price for a long and uncertain time, the effects of a shock are spread over an indefinite horizon. This tends to make aggregate inflation more persistent. In the Taylor model, by contrast, the price chosen at time $t$ is known to expire after $N$ periods, so the effect of a shock is transmitted only through a finite number of cohorts and disappears more quickly. For this reason, price adjustment is in general less persistent than in the Calvo model.

The two models also differ in tractability. The Calvo model is usually easier to handle analytically because its probabilistic structure leads to a recursive structure. The Taylor model is more transparent in terms of timing, since price durations are fixed and firms can be grouped into $N$ cohorts, but it is less convenient analytically because one must keep track of those cohorts explicitly. 
                
                
            \end{solution}
    
        \item Recalling that the aggregate price index \( P_t \) is defined by:
        \[
        P_t = \left( \int_0^1 P_t(i)^{1 - \epsilon} \, di \right)^{\frac{1}{1 - \epsilon}},
        \]
        show that, around a zero-inflation steady state, the (log) price level will satisfy:
        \[
        p_t = \frac{1}{N} \sum_{k=0}^{N-1} p_{t-k}^*.
        \]
    
        \textit{[Hint: Argue why we have symmetric pricing for firms within each cohort and derive the relationship between the log price level \( p_t \) and the optimal reset prices \( p_{t-k}^* \).]}
    
            \begin{solution}
In the Taylor model, at every period $t$, only a fraction $1/N$ of firms reset their price. Equivalently, in the remaining firms that didn't set the price at time $t$, we have a fraction $1/N$ that have reset it at $t-1$, and so on.  Now, since firms that reset the price in the same period optimizes through the same first order condition (which is not firm-specific), and applying the Law of Large numbers, we can write:

\begin{align*}
    P_t^{1-\varepsilon}&=\frac{1}{N}(P_t^*)^{1-\varepsilon}+\frac{1}{N} (P_{t-1}^*)^{1-\varepsilon} + \dots + \frac{1}{N} (P_{t-(N-1)}^*)^{1-\varepsilon} \\
    &= \frac{1}{N} \sum_{k=0}^{N-1}(P_{t-k}^*)^{1-\varepsilon}
\end{align*}

We can now express this relation as a function of log-deviations from a zero-inflation steady state:
\begin{align*}
    P^{1-\varepsilon}e^{(1-\varepsilon)\hat p_t}=\frac{1}{N} \sum_{k=0}^{N-1}P^{1-\varepsilon}e^{(1-\varepsilon)\hat p_{t-k}^*}.
\end{align*}
Taking the first order Taylor expansion around the zero-inflation steady state:
\begin{align*}
    1+(1-\varepsilon)\hat p_t \approx \frac{1}{N}\sum_{k=0}^{N-1} (1+(1-\varepsilon)\hat p_{t-k}^*).
\end{align*}
Simplifying, we get:
\[
\hat p_t=\frac{1}{N}\sum_{k=0}^{N-1}\hat p_{t-k}^* \implies p_t-\log P = \frac{1}{N}\sum_{k=0}^{N-1}(p_{t-k}^*-\log P) \implies p_t=\frac{1}{N}\sum_{k=0}^{N-1}p_{t-k}^*,
\]
that is the desired expression. 
                
            \end{solution}
    
    \end{enumerate}

\section{The Phillips Curve with Price Indexation}

    In the New Keynesian model, there is a probability $\theta$ that a firm will not be able to change its nominal price in a given time period. Firms that do have an opportunity to change price at time $t$ choose a reset price $p^*_t$ according to:
    \[
    p^*_t = (1 - \beta \theta) \sum_{\ell=0}^\infty (\beta \theta)^\ell \mathbb{E}_t [p_{t+\ell} + \gamma \widehat{mc}_{t + \ell }],
    \]
    where $\beta$ is the discount factor, $p_t$ is the price level, $\widehat{mc}_t$ is average marginal cost, and $\gamma = \frac{1 - \alpha}{1 - \alpha + \alpha \epsilon}$ is a positive coefficient. All variables are given as log deviations from their steady-state values. The general price level $p_t$ is determined by:
    \[
    p_t = (1 - \theta) \sum_{\ell=0}^\infty \theta^\ell p^*_{t-\ell}.
    \]

    \begin{enumerate}
    
        \item Provide an intuitive explanation of the two equations above.
    
            \begin{solution}
    The first equation describes the optimal reset price for the firm:
    \[
    p^*_t = (1 - \beta \theta) \sum_{\ell=0}^\infty (\beta \theta)^\ell \mathbb{E}_t [p_{t+\ell} + \gamma \widehat{mc}_{t + \ell }].
    \]
    In particular, we can see that the equation is clearly forward-looking. The intuition is the following: since firms know that they could not be able to change their price in the future, they consider also their optimal future prices. Moreover, firms also consider the future marginal costs. Hence, the optimal today reset price is a weighted average of future prices and marginal costs, with weights that sum up to one:
    \[
    (1-\beta \theta)\sum_{\ell =0}^\infty (\beta \theta) ^\ell = (1-\beta \theta)\frac{1}{1-\beta \theta}=1.
    \]
    We can see that:
    \begin{itemize}
        \item If firms expect higher prices in the future $p_{t+\ell}$, then they want to raise prices today, otherwise they risk to remain stucked at a too low price.
        \item If firms expect higher marginal costs in the future $\widehat{mc}_{t+\ell}$, then they want to raise the reset price today, because the product is expected to be more expensive tomorrow.
        \item $\gamma$ tells us how strongly today reset price reacts to future marginal cost.
        \item $\theta^\ell$ is the probability that today's price will remain the same until period $\ell$. 
    \end{itemize}
    The second equation is just the aggregate price level:
    \[
    p_t = (1 - \theta) \sum_{\ell=0}^\infty \theta^\ell p^*_{t-\ell}.
    \]
    As we can see, today aggregate price level is simply the weighted average of current and past reset prices. Following the intuition of Exercise 1, we have:
    \begin{itemize}
        \item a fraction of firms $1-\theta$ that resets the price today,
        \item a fraction of firms $(1-\theta)\theta$ that have resetted the price yesterday,
        \item a fraction of firms $(1-\theta)\theta^2$ that have resetted the price two periods ago, and so on.
    \end{itemize}
    
    
                
            \end{solution}
    
        \item Show that the reset-price and price-level equations are equivalent to:
        \[
        p^*_t = \beta \theta \mathbb{E}_t p^*_{t+1} + (1 - \beta \theta)(p_t +  \gamma \widehat{mc}_{t}),
        \]
        and
        \[
        p_t = \theta p_{t-1} + (1 - \theta) p^*_t.
        \]
    
            \begin{solution}
 Take the optimal reset-price equation:   
\[
    p^*_t = (1 - \beta \theta) \sum_{\ell=0}^\infty (\beta \theta)^\ell \mathbb{E}_t [p_{t+\ell} + \gamma \widehat{mc}_{t + \ell }].
    \]
Then, iterate it 1-period ahead:
\[
    p^*_{t+1} = (1 - \beta \theta) \sum_{\ell=0}^\infty (\beta \theta)^\ell \mathbb{E}_{t+1} [p_{t+1+\ell} + \gamma \widehat{mc}_{t+1 + \ell }],
    \]
and note that, by the general law of iterated expectations:
\begin{align*}
\mathbb{E}_t p^*_{t+1} &= \mathbb{E}_t (1 - \beta \theta) \sum_{\ell=0}^\infty (\beta \theta)^\ell \mathbb{E}_{t+1} [p_{t+1+\ell} + \gamma \widehat{mc}_{t+1 + \ell }] \\
&= (1 - \beta \theta) \sum_{\ell=0}^\infty (\beta \theta)^\ell \mathbb{E}_{t} [p_{t+1+\ell} + \gamma \widehat{mc}_{t+1 + \ell }].
\end{align*}
Now, we can go back to the optimal reset price today:
\begin{align*}
    p^*_t &= (1 - \beta \theta) \sum_{\ell=0}^\infty (\beta \theta)^\ell \mathbb{E}_t [p_{t+\ell} + \gamma \widehat{mc}_{t + \ell }] \\
    &= (1-\beta \theta) (p_t+\gamma \widehat{mc}_t)+(1-\beta \theta) \sum_{\ell=1}^\infty (\beta \theta)^\ell \mathbb{E}_t[p_{t+\ell}+\gamma \widehat{mc}_{t+\ell}] \\
    &= (1-\beta \theta)(p_t+\gamma \widehat{mc}_t)+(1-\beta\theta)\sum_{\ell=0}^\infty (\beta\theta)^{\ell+1}\mathbb{E}_t[p_{t+1+\ell}+\gamma \widehat{mc}_{t+1+\ell}] \\
    &=(1-\beta \theta)(p_t+\gamma \widehat{mc}_t)+(\beta \theta) (1-\beta\theta)\sum_{\ell=0}^\infty (\beta\theta)^{\ell}\mathbb{E}_t[p_{t+1+\ell}+\gamma \widehat{mc}_{t+1+\ell}] \\
    &= (1-\beta \theta)(p_t+\gamma \widehat{mc}_t) + \beta \theta \mathbb{E}_t p_{t+1}^*,
\end{align*}
that is the desired expression.

Similarly, take the price-level equation for period $t-1$:
\[
p_{t-1}=(1-\theta)\sum_{\ell=0}^\infty \theta^\ell p_{t-1-\ell}^*.
\]
Then:
\begin{align*}
    p_{t}&=(1-\theta)\sum_{\ell=0}^\infty \theta^\ell p_{t-\ell}^* \\
    &= (1-\theta) p_t^*+(1-\theta) \sum_{\ell=1}^\infty \theta^\ell p_{t-\ell}^* \\
    &= (1-\theta)p_t^*+(1-\theta)\sum_{\ell=0}^\infty \theta^{\ell+1}p_{t-1-\ell}^* \\
    &= (1-\theta)p_t^*+\theta (1-\theta)\sum_{\ell=0}^\infty \theta^{\ell}p_{t-1-\ell}^* \\
    &= (1-\theta)p_t^*+\theta p_{t-1}.
\end{align*}
                
                
            \end{solution}
    
        \item By combining the two equations, derive the New Keynesian Phillips curve for inflation $\pi_t = p_t - p_{t-1}$:
        \[
        \pi_t = \beta \mathbb{E}_t \pi_{t+1} + \kappa \widetilde{y}_t,
        \]
        using that $\widehat{mc}_t = \left( \sigma + \frac{\alpha + \phi}{1 - \alpha}  \right) \widetilde{y}_t$, and where $\kappa = \gamma \left( \sigma + \frac{\alpha + \phi}{1 - \alpha}  \right) \left( \frac{(1 - \theta)(1 - \beta \theta)}{\theta}    \right)$
    
            \begin{solution}
First, notice that:
\begin{align*}
    p_t=(1-\theta)p_t^* +\theta p_{t-1} \implies p_t^*=\frac{p_t-\theta p_{t-1}}{1-\theta},
\end{align*}
and, similarly:
\begin{align*}
    \mathbb{E}_tp_{t+1}=(1-\theta)\mathbb{E}_tp_{t+1}^* +\theta p_{t} \implies \mathbb{E}_tp_{t+1}^*=\frac{\mathbb{E}_tp_{t+1}-\theta p_{t}}{1-\theta}
\end{align*}
Therefore, the optimal reset price equation becomes:
\begin{align*}
    \frac{p_t-\theta p_{t-1}}{1-\theta}=(1-\beta\theta)(p_t+\gamma \widehat{mc}_t)+\beta \theta \frac{ \mathbb{E}_tp_{t+1}-\theta p_t}{1-\theta}.
\end{align*}
Now, notice that:
\begin{align*}
    \pi_t= p_t-p_{t-1} \iff p_{t-1}=p_t-\pi_t,
\end{align*}
and
\begin{align*}
    \mathbb{E}_t\pi_{t+1}=\mathbb{E}_t p_{t+1}-p_t \iff \mathbb{E}_tp_{t+1}=p_t+\mathbb{E}_t\pi_{t+1}.
\end{align*}
Hence:
\begin{align*}
    p_t-\theta p_{t-1}=(1-\theta)(1-\beta \theta)(p_t+\gamma \widehat{mc}_t)+\beta \theta (\mathbb{E}_t p_{t+1}-\theta p_t)
\end{align*}
that is:
\begin{align*}
    p_t-\theta(p_t-\pi_t)&=(1-\theta)(1-\beta \theta)(p_t + \gamma \widehat{mc}_t)+\beta \theta (\mathbb{E}_t \pi_{t+1}+p_t-\theta p_t) \\
    (1-\theta)p_t+\theta \pi_t&=(1-\theta)(1-\beta \theta)p_t + (1-\theta)(1-\beta \theta)\gamma \widehat{mc}_t+\beta \theta \mathbb{E}_t \pi_{t+1}+\beta \theta(1-\theta)p_t \\
    \theta\pi_t&=(1-\theta)(1-\beta \theta) \gamma \widehat{mc}_t +\beta \theta \mathbb{E}_t \pi_{t+1} \\
    \pi_t&=\beta \mathbb{E}_t \pi_{t+1}+\frac{(1-\theta)(1-\beta \theta)}{\theta}\gamma \left( \sigma + \frac{\alpha+\phi}{1-\alpha} \right) \tilde{y}_t \\
    \pi_t &= \beta \mathbb{E}_t \pi_{t+1}+ \kappa \tilde{y}_t,
\end{align*}
that is the desired expression.
                
            \end{solution}
            
    \end{enumerate}

    \noindent Next suppose that when a firm does not have an opportunity to choose a new reset price optimally at time $t$ (probability $\theta$), the firm adjusts its old nominal price in line with any past inflation $\pi_{t-1}$ that has occurred.\footnote{For a more detailed treatment, see for example "Trend Inflation, Indexation, and Inflation Persistence in the New Keynesian Phillips Curve" by Cogley and Sbordone (\href{https://www.aeaweb.org/articles?id=10.1257/aer.98.5.2101}{AER 2008}).} This means the equation for the price level is now:
    \[
    p_t = \theta (p_{t-1} + \pi_{t-1}) + (1 - \theta) p^*_t.
    \]
    
    \noindent When choosing the reset price $p^*_t$, firms take account of the indexation of prices to past inflation. Hence the new equation for the reset price is:
    \[
    p^*_t = (1 - \beta \theta) \sum_{\ell=0}^\infty (\beta \theta)^\ell \mathbb{E}_t [p_{t+\ell} - (p_{t+\ell-1} - p_{t-1}) +  \gamma \widehat{mc}_{t+\ell}].
    \]

    \begin{enumerate}[resume]

        \item Explain the intuition for the new reset-price equation.

            \begin{solution}
As we said before, without indexation, a firm that resets the price today chooses a  price knowing that it may remain stuck with that same price for several periods. Therefore, it sets $p_t^*$ as a weighted average of te future prices it would like to charge $p_{t+\ell}+\gamma \widehat{mc}_{t+\ell}$. With indexation, instead, if the firm cannot reset today, its price gets automatically updated with inflation. Therefore, the firm has to take into account this fact and the reset price equation adds an extra term, namely $p_{t+\ell-1}-p_{t-1}$, that is the cumulative indexation between $t$ and $t+\ell$. Indeed, notice that:
\begin{align*}
    \pi_t+ \pi_{t+1}+ \dots + \pi_{t+\ell-1}&= p_t-p_{t-1}+ p_{t+1}-p_t + \dots +p_{t+\ell-1}-p_{t+\ell-2} \\
    &= p_{t+\ell-1}-p_{t-1}.
\end{align*}
            
            
            \end{solution}

        \item Derive the new Phillips curve. Comment on any differences relative to the standard New Keynesian Phillips curve. 

            \textit{[Hint: Start by putting the reset-price equation in a simpler recursive form.]} 

            \begin{solution}
Equivalently to what we did before, let's write:
\begin{align*}
     p^*_t &= (1 - \beta \theta) \sum_{\ell=0}^\infty (\beta \theta)^\ell \mathbb{E}_t [p_{t+\ell} - (p_{t+\ell-1} - p_{t-1}) +  \gamma \widehat{mc}_{t+\ell}] \\
     &=(1-\beta \theta) (p_t+\gamma \widehat{mc}_t)+(1-\beta \theta) \sum_{\ell =1}^\infty (\beta \theta)^\ell  \mathbb{E}_t [p_{t+\ell} - (p_{t+\ell-1} - p_{t-1}) +  \gamma \widehat{mc}_{t+\ell}] \\
     &=(1-\beta \theta) (p_t+\gamma \widehat{mc}_t)+(1-\beta \theta) \sum_{\ell =0}^\infty (\beta \theta)^{\ell+1}  \mathbb{E}_t [p_{t+\ell+1} - (p_{t+\ell} - p_{t-1}) +  \gamma \widehat{mc}_{t+\ell+1}] \\
     &=(1-\beta \theta) (p_t+\gamma \widehat{mc}_t)-\beta \theta(p_t-p_{t-1})+\beta \theta (1-\beta \theta) \sum_{\ell =0}^\infty (\beta \theta)^{\ell}  \mathbb{E}_t [p_{t+\ell+1} - (p_{t+\ell} - p_{t}) +  \gamma \widehat{mc}_{t+\ell+1}] \\
     &=(1-\beta \theta)(p_t+\gamma\widehat{mc}_t)-\beta \theta\pi_t +\beta \theta \mathbb{E}_t p_{t+1}^*.
\end{align*}
From the price level equation, we know that:
\begin{align*}
    p_t^* =\frac{p_t-\theta(p_{t-1}+\pi_{t-1})}{1-\theta},
\end{align*}
and, one period ahead:
\[
\mathbb{E}_tp_{t+1}^* =\frac{\mathbb{E}_tp_{t+1}-\theta(p_{t}+\pi_{t})}{1-\theta}.
\]
Therefore:
\begin{align*}
    \frac{p_t-\theta(p_{t-1}+\pi_{t-1})}{1-\theta}& = (1-\beta \theta)(p_t+\gamma \widehat{mc}_t)-\beta \theta\pi_t +\beta \theta \frac{\mathbb{E}_tp_{t+1}-\theta(p_{t}+\pi_{t})}{1-\theta} \\
    p_t-\theta(p_{t-1}+\pi_{t-1})&=(1-\theta)(1-\beta \theta)(p_t+\gamma \widehat{mc}_t)-\beta \theta (1-\theta) \pi_t +\beta \theta (\mathbb{E}_tp_{t+1}-\theta(p_{t}+\pi_{t}))
\end{align*}
Recall that:
\begin{align*}
    p_{t-1}&=p_t-\pi_t, \\
    \mathbb{E}_t p_{t+1}&= p_t+\mathbb{E}_t \pi_{t+1}.
\end{align*}
Then:
\begin{align*}
    p_t-\theta(p_t-\pi_t+\pi_{t-1})&=(1-\theta)(1-\beta\theta)p_t +(1-\theta)(1-\beta\theta) \gamma \widehat{mc}_t\\&-\beta \theta(1-\theta) \pi_t +\beta \theta (p_t+\mathbb{E}_t\pi_{t+1}-\theta (p_t+\pi_t)) \\
    (1-\theta) p_t +\theta(\pi_t-\pi_{t-1})&= (1-\theta)(1-\beta\theta)p_t +(1-\theta)(1-\beta\theta)\gamma \widehat{mc}_t\\&-\beta \theta(1-\theta) \pi_t+\beta \theta (1-\theta)p_t+\beta \theta (\mathbb{E}_t \pi_{t+1}-\theta \pi_t) \\
    \theta (\pi_t-\pi_{t-1}) &= (1-\theta)(1-\beta\theta)\gamma  \widehat{mc}_t -\beta \theta(1-\theta) \pi_t +\beta \theta (\mathbb{E}_t\pi_{t+1}-\theta \pi_t) \\
    \pi_t-\pi_{t-1} &= \kappa \tilde{y}_t -\beta(1-\theta)\pi_t+\beta \mathbb{E}_t \pi_{t+1}-\beta \theta \pi_t \\
    \pi_t -\pi_{t-1}&= \kappa \tilde{y}_t +\beta (\mathbb{E}_t \pi_{t+1}-\pi_t)
\end{align*}
or, equivalently:
\begin{align*}
    \pi_t= \frac{1}{1+\beta}\pi_{t-1}+\frac{\beta}{1+\beta} \mathbb{E}_t \pi_{t+1}+\frac{\kappa}{1+\beta}\tilde{y}_t
\end{align*}
The main difference with the standard NK Phillips curve is the presence of a backward looking term: $\pi_{t-1}$. Hence, inflation becomes intrinsically more persistent: even if marginal cost falls today, inflation does not jump down immediately, because part of pricing is mechanically tied to last period’s inflation.

Moreover, the forward looking inflation component is still here, $\mathbb{E}_t\pi_{t+1}$, but the coefficient in front of it is different:
\[
\frac{\beta}{1+\beta}< \beta. 
\]
In other words, the forward looking component matters less. Moreover, also the slope of the PC is lower:
\[
\frac{\kappa}{1+\beta}<\kappa.
\]
            
            
        \end{solution}

    \end{enumerate}

            \item Read and summarize the key question and findings of "Is the Phillips Curve Alive and Well after All? Inflation Expectations and the Missing Disinflation" by Coibion and Gorodnichenko (\href{https://www.aeaweb.org/articles?id=10.1257/mac.20130306}{AEJ: Macro 2015}). 

                \begin{solution}

                    The central question addressed by Coibion \& Gorodnichenko (2015) is the ``missing disinflation'' puzzle: why did US inflation not fall much further (or enter deflation) during the Great Recession of 2008--2011, despite the massive spike in unemployment and economic slack? Traditional Phillips curve models, which relied on backward-looking expectations or forecasts by professionals, predicted a much sharper disinflation.

                    Their key finding is that the Phillips curve is actually ``alive and well'' once you use the \emph{correct} measure of inflation expectations. They document that household inflation expectations (as measured by the Michigan Survey of Consumers) closely track salient prices, particularly oil and gasoline. During the 2009--2011 period, a surge in global oil prices pushed household inflation expectations higher, even as the broader economy remained in a severe slump. When these survey-based expectations are used in an expectations-augmented Phillips curve, the implied negative inflation surprise ($\pi_t - E^{MSC}_t$) matches the high unemployment rate perfectly. This fully explains the missing disinflation without needing to assume that the underlying slope of the Phillips curve had flattened.

                    A complementary piece of evidence underpins the entire interpretation. \textcite{coibion2015phillips} exploit the New Zealand business survey---one of the few datasets that directly measures \emph{firm} inflation expectations---and document that firms' expectations co-move strongly with household survey forecasts and only weakly with professional forecasts. This finding is what licenses the use of MSC data as a proxy for the expectations relevant to firm price-setting, and is therefore central to the paper's claim that the Phillips curve is ``alive and well'' once the right expectations measure is used.
                    
                \end{solution}

                    A fundamental identification problem afflicts all the OLS regressions estimated above (in Questions 2, 3, and 6): \textbf{simultaneity (reverse causality)}. Inflation expectations and unemployment are \emph{jointly determined}---aggregate demand shocks simultaneously raise inflation (or expectations thereof) and lower unemployment, while monetary policy responses to inflation and employment conditions shift the economy along both dimensions. OLS of inflation on unemployment therefore conflates movements \emph{along} the Phillips curve (driven by supply shocks, which we want to identify) with shifts \emph{of} the Phillips curve (driven by demand shocks and policy responses).

They show that when the central bank conducts \emph{optimal monetary policy}---adjusting interest rates to stabilise inflation in response to unemployment gaps---the policy reaction function induces a spurious positive correlation between inflation and unemployment in equilibrium. This endogeneity biases the OLS estimate of the Phillips curve slope toward zero (or even toward a positive sign). The PC is identified \emph{only} off supply-side shocks or when an instrumental variable for monetary policy is available.

    \subsection{Non-linear Phillips curves}

        \begin{enumerate}[resume]
            \item Read ``\textit{The Slanted-L Phillips Curve}`` by Benigno and Eggertsson (2024) (\href{https://doi.org/10.1257/pandp.20241051}{AEA Papers and Proceedings, 2024}). Briefly summarize the authors’ key arguments.

                \begin{solution}

                    Benigno \& Eggertsson (2024) argue that the standard \emph{linear} New Keynesian Phillips curve misses a critical feature of the inflation--labour-market-tightness relationship: it is \textbf{non-linear, exhibiting a convex (or kinked) shape}. Specifically, they document a ``slanted-L'' pattern in the data:

                    \begin{itemize}
                        \item \textbf{At low tightness} (high unemployment, low $V/U$), additional slack has \emph{little} disinflationary effect---the Phillips curve is nearly flat. This explains the ``missing disinflation'' of 2009--2015: despite very high unemployment, inflation did not fall substantially because the economy was on the flat portion of the curve.
                            
                        \item \textbf{At high tightness} (low unemployment, high $V/U$), small increases in tightness generate \emph{large} inflation increases---the curve becomes steep, nearly vertical. This explains the rapid inflation surge of 2021--2022: the economy quickly moved onto the steep portion of the curve.
                    \end{itemize}
                    
                    The key theoretical mechanism is a \emph{capacity constraint on production}: firms can substitute between vacancies and labour to adjust output, but as the labour market approaches full utilisation the marginal cost curve becomes very steep---effectively kinking---because additional output requires either bidding scarce workers away from other firms or accepting longer vacancy durations. Once this constraint binds, small increases in tightness map into large increases in marginal cost and hence in price-setting markups, generating the near-vertical arm of the slanted-L. Below the kink, marginal cost is roughly flat in tightness, so additional slack has little disinflationary bite. Search-and-matching frictions are a complementary force: when $V/U$ is high, hiring takes longer and wage offers must rise to attract workers, reinforcing the cost-side amplification. Crucially, Benigno \& Eggertsson argue that \emph{no structural break or shift in the Phillips curve is needed} to explain both the ``missing disinflation'' and the ``inflation surge''---both phenomena arise naturally from the non-linear shape of a \emph{stable} underlying relationship.
                    
                \end{solution}

                        \includegraphics[trim=0cm 0cm 0cm 1cm, clip, width=0.75\linewidth]{images/q8-nonlinear-1.pdf}
                        \caption{The non-linear Phillips curve.}
                        \label{fig:q8-nonlinear-1}
                    \end{figure}

                    The scatterplot reveals the characteristic \textbf{``slanted-L'' shape} described by \textcite{benigno2024slanted}. For $V/U$ ratios below roughly $1$, the relationship is essentially flat: across the 2000--2019 expansion, the GFC trough, and the late-cycle 2018--2019 quarters where $V/U$ first crossed above one with inflation still anchored around 2\%, year-over-year PCE inflation hovers in a narrow $1$--$2.5\%$ band regardless of tightness. The steep, near-vertical arm of the curve is essentially the \textbf{2021:Q2--2023:Q1 cluster}: every quarter in our sample with PCE inflation above 4\% sits in this window, with $V/U$ ratios between $1.04$ and $1.93$. Within this cluster a roughly $0.9$-point increase in $V/U$ (from $1.04$ to $1.93$ over 2021:Q2--2022:Q2) is associated with PCE inflation rising from $4.0\%$ to $6.9\%$. The kink itself sits near $V/U \approx 1$---close to where \textcite{benigno2024slanted} place it in their Figure~1.

                    This pattern contrasts sharply with the \textbf{linear Phillips curve}, which would predict a uniform, moderate negative relationship between slack and inflation across the entire range. The linear specification would substantially overpredict disinflation during the slack post-GFC period and underpredict inflation during the 2021--2022 boom. The non-linear relationship shows that inflation is relatively insensitive to tightness when the labour market has ample slack---firms can fill vacancies without large wage increases---but becomes highly sensitive once tightness pushes the economy past the kink and onto the capacity-constrained arm. As the post-2022 observations show, the unwinding of tightness brought inflation back down, tracing the curve in reverse: the 2024--2025 quarters are back at $V/U\approx 1$ with PCE inflation around $2.5\%$.
                    
                \end{solution}

            \item Lastly, read Giulia Gitti's Job Market Paper (\href{https://drive.google.com/file/d/1ZD9WUJ06xZFO8bhZsQyy3tZZfk2yC8oH/view}{JMP, 2024}). Explain her approach of addressing potential endogeneity when relating labor market tightness and inflation. How does this relate to your findings in 3.8 and 3.9?

                \begin{solution}

                    \textcite{gitti2024nonlinearities} addresses a key endogeneity concern that plagues the OLS scatterplot in Question~9: the vacancy--unemployment ratio $V/U$ is endogenous to inflation because demand shocks raise both labour-market tightness and prices simultaneously. Her identification strategy uses a Bartik-style (shift-share) instrumental variable in which the ``shifts'' are national-level vacancy shocks in \emph{tradable intermediate-input} industries---chosen because demand for these industries' output is determined by national rather than local conditions---and the ``shares'' are pre-determined local industry employment weights. Interacting the two yields exogenous variation in local labour-market tightness that is plausibly orthogonal to local demand shocks. To further isolate demand-driven variation, she controls for the relative price of intermediate inputs and for productivity shocks in the final-goods sector.
                    
                    This approach relates directly to Question 9 (and Question 7): the OLS scatterplot is subject to the \textbf{same simultaneity bias} discussed previously. During demand booms, both tightness and inflation rise \emph{simultaneously}, inflating the apparent slope of the tightness--inflation relationship. Conversely, during recessions, both decline together. The OLS estimates therefore confound the causal effect of tightness on inflation with the common response of both variables to demand conditions.
                    
                    Gitti's IV strategy recovers a \textbf{cleaner causal estimate} of the tightness--inflation relationship, which she finds to be:
                    
                    \begin{enumerate}
                        \item \textbf{Non-linear}, consistent with the slanted-L shape documented by Benigno \& Eggertsson (2024);
                        \item \textbf{Quantitatively different} from OLS---the IV estimates suggest the true causal relationship is steeper in the tight-market region than OLS implies (because OLS averages over demand- and supply-driven variation);
                        \item \textbf{Robust} to controlling for local demand conditions and to various specification checks on the instrument's validity.
                    \end{enumerate}
                    
                    Her results thus provide rigorous causal support for the non-linear Phillips curve hypothesis and demonstrate that naive OLS estimates---whether using unemployment (Q2, Q5) or tightness (Q8)---are biased by simultaneity.
                    
                \end{solution}
                
        \end{enumerate}
